FROM python:3.10-slim

WORKDIR /app

# Instala ferramentas básicas do sistema
RUN apt-get update && apt-get install -y gcc libpq-dev && rm -rf /var/lib/apt/lists/*

# Copia e instala as dependências do Python (incluindo o gunicorn)
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copia todos os arquivos do site para dentro do servidor da nuvem
COPY . .

# Libera a porta 8080 exigida pelo Google Cloud Run
EXPOSE 8080

# Comando que inicia o site usando o Gunicorn na porta correta
CMD ["gunicorn", "--bind", "0.0.0.0:8080", "--workers", "1", "--threads", "8", "--timeout", "0", "app:app"]
